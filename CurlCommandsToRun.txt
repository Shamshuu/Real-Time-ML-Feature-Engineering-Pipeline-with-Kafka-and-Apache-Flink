docker compose up --build -d

Verify Infrastructure Health:

curl -s http://localhost:8081/overview

curl -s http://localhost:8501/_stcore/health


Verify Flink Job Submission:

#List all Flink jobs and format the output

curl -s http://localhost:8081/jobs | grep -o '"id":"[^"]*","status":"RUNNING"'

#Extract the RUNNING Job ID into a variable for later use


JOB_ID=$(curl -s http://localhost:8081/jobs | grep -o '"id":"[^"]*","status":"RUNNING"' | cut -d'"' -f4)
echo "My Flink Job ID is: $JOB_ID"


Observe the Data Flow:

#Verify topics were created automatically

docker exec real-time-ml-feature-engineering-pipeline-with-kafka-and-apache-flink-kafka-1 \
  kafka-topics --bootstrap-server localhost:9092 --list


#View the RAW input stream (Events generated by the Python Producer)
#This shows the simulated user clicks, views, etc.

docker exec real-time-ml-feature-engineering-pipeline-with-kafka-and-apache-flink-kafka-1 \
  kafka-console-consumer --bootstrap-server localhost:9092 --topic user-events --max-messages 3

#View the COMPUTED FEATURES output (The results from Flink)
#Wait a few more seconds if this is empty, as the 1-hour windows (simulated in 1 minute) need to tumble.

docker exec real-time-ml-feature-engineering-pipeline-with-kafka-and-apache-flink-kafka-1 \
  kafka-console-consumer --bootstrap-server localhost:9092 --topic feature-store \
  --from-beginning --max-messages 5 --property print.key=true
